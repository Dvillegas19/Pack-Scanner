<script>
  import {onMount} from "svelte";
  import {supabase} from "$lib/supabaseClient"; 

  let operarios=[];
  let ordenes=[];
  let consumibles=[];
  let solicitudes=[];
  let loading=true

  onMount(async()=>{
    loading=true;
    await Promise.all([
    fetchOperarios(),
    fetchOrdenes(),
    fetchConsumibles(),
    fetchSolicitudes(),
    ]);
    loading=false;
  });

async function fetchOperarios() {
  const {data,error}=await supabase
  .from("usuario")
  .select("id, nombre")

  if (error){
      console.error("Error cargando operarios:",error)
  } else {
      operarios=data
  }
}

async function fetchSolicitudes() {
    const {data,error}=await supabase
    .from("solicitud_consumible")
    .select("codigo_solicitud, orden_id, fecha_entrega, usuario(nombre)")
  if(error){
      console.error("Error cargando solicitudes:", error)
  } else {
    solicitudes=data.map((solicitud) =>(
      {codigo: solicitud.codigo_solicitud,
      orden_id: solicitud.orden_id,
      operario: solicitud.usuario.nombre,
      fecha: solicitud.fecha_entrega}
    ));
  }
}

async function fetchOrdenes() {
    const {data,error}=await supabase
    .from("orden_produccion")
    .select("codigo_orden, fecha_emision")
    if (error){
     console.error("Error cargando ordenes:", error)
    } else{
    ordenes=data
  }
}

async function fetchConsumibles(){
    const {data,error}=await supabase
    .from("consumible")
    .select("codigo_barra, nombre, numero_bodega")
  if (error){
    console.error("Error cargando consumibles:", error)
  } else{
    consumibles=data
  }
}

let selectedOperario='';
let selectedOrden='';
let selectedFecha=new Date().toISOString().split('T')[0];
let filtroFecha='';
let especificaciones=[];

let selectConsumibles=[
  {consumible_id:'',cantidad_requerida: 1}
];

function addConsumible(){
  selectConsumibles=[...selectConsumibles,{consumible_id:'',cantidad_requerida: 1}];
}

function deleteConsumible(index){ 
  selectConsumibles=selectConsumibles.filter((_,i)=> i !== index);
}

async function enviarSolicitud() {  
  if(!selectedOperario||!selectedOrden||selectConsumibles.length==0){
    alert ("Por favor complete todos los campos");
    return;
  }
  const {data: solicitudData, error:solicitudError} = await supabase
  .from("solicitud_consumible")
  .insert({
    orden_id:selectedOrden,
    operario_id:selectedOperario,
    fecha_entrega:selectedFecha
  })
  .select("codigo_solicitud")
  .single();

  if (solicitudError){
    console.error("Error creando la solicitud:",solicitudError);
    alert ("Error creando la solicitud");
    return;
  }
  const nuevaSolicitudId=solicitudData.codigo_solicitud;
  
  const especificacionesConsumibles=selectConsumibles
  .filter(consumibles=>consumibles.consumible_id && consumibles.cantidad_requerida>0)
  .map(consumibles=>({
    solicitud_id:nuevaSolicitudId,
    consumible_id:consumibles.consumible_id,
    cantidad_requerida:consumibles.cantidad_requerida,
  }));
  if (especificacionesConsumibles.length==0){
    alert("Debes añadir al menos un consumible válido");
    return;
  }
  const {error:especificacionesError}=await supabase
  .from("especificacion")
  .insert(especificacionesConsumibles);
  if (especificacionesError){
    console.error("Error insertando especificaciones:",especificacionesError);
    alert ("Error guardando los consumibles de la solicitud")
    return;
  }
  alert("¡Solicitud creada con éxito!")
  selectedOperario="";
  selectedOrden="";
  selectConsumibles=[{consumible_id:'',cantidad_requerida:1}];
  await fetchSolicitudes()
}

async function filtrarFecha(){
  if(!filtroFecha){
    alert("Debe poner una fecha para filtrar"); 
    return;
  }
  loading=true;
  const{data, error} =await supabase
  .from("solicitud_consumible")
  .select("codigo_solicitud, orden_id, fecha_entrega, usuario(nombre)")
  .eq("fecha_entrega",filtroFecha)
  if(error){
    console.error("Error filtrando las solicitudes",error);
  } else{
    solicitudes=data.map((solicitud)=>({
      codigo:solicitud.codigo_solicitud,
      orden:solicitud.orden_id,
      operario:solicitud.usuario.nombre,
      fecha:solicitud.fecha_entrega
    }));
  }
}
async function limpiarFiltro() {
  filtroFecha="";
  loading=true;
  await fetchSolicitudes();
  loading=false;
}

async function verDetalle(codigo_solicitud){
  const {data, error}= await supabase
  .from("especificacion")
  .select("*")
  .eq("solicitud_id",codigo_solicitud)
  if (error){
    console.error("Error viendo detalles",error);
  } else{
      especificaciones=data;
  }
}
