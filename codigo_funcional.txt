<script>
  import { onMount } from 'svelte';
  import { supabase } from '$lib/supabaseClient';

  // --- 1. Variables para mostrar datos ---
  let operarios = [];
  let ordenes = [];
  let consumibles = [];
  let solicitudes = [];
  let loading = true;

  // --- 2. Variables para el Formulario ---
  let selectedOperario = '';
  let selectedOrden = '';
  let selectedFecha = new Date().toISOString().split('T')[0];
  let filtroFecha = '';

  let items = [{ consumible_id: '', cantidad_requerida: 1 }];

  // --- HU-012: Variables para el filtro de consumibles ---
  let consumibleSearchTerm = '';
  
  // --- ¡CAMBIO 1 (Arregla Bug Visual)! ---
  // Ahora, la lista filtrada depende de 'consumibleSearchTerm' Y de 'consumibles'
  // Se actualizará si CUALQUIERA de las dos cambia.
  $: filteredConsumibles = getFilteredConsumibles(consumibleSearchTerm, consumibles);
  
  // --- HU-014: Variable de estado para Edición ---
  let editingId = null; 

  // --- LÓGICA DE CARGA INICIAL ---
  onMount(async () => {
    loading = true;
    await Promise.all([
      fetchOperarios(),
      fetchOrdenes(),
      fetchConsumibles(),
      fetchSolicitudes(),
    ]);
    loading = false;
  });

  // --- FUNCIONES FETCH (Cargar datos) ---
  async function fetchOperarios() {
    const { data, error } = await supabase.from('usuario').select('id, nombre');
    if (error) console.error('Error cargando operarios:', error);
    else operarios = data;
  }

  async function fetchSolicitudes() {
    const { data, error } = await supabase
      .from('solicitud_consumible')
      .select('codigo_solicitud, orden_id, fecha_entrega, operario_id, usuario ( nombre )');
    if (error) {
      console.error('Error cargando solicitudes:', error);
    } else {
      solicitudes = data.map((solicitud) => ({
        codigo: solicitud.codigo_solicitud,
        orden_id: solicitud.orden_id,
        operario: solicitud.usuario.nombre,
        operario_id: solicitud.operario_id,
        fecha: solicitud.fecha_entrega,
        expanded: false,
        loadingDetails: false,
        details: []
      }));
    }
  }

  async function fetchOrdenes() {
    const { data, error } = await supabase
      .from('orden_produccion')
      .select('codigo_orden, fecha_emision');
    if (error) console.error('Error cargando ordenes:', error);
    else ordenes = data;
  }

  async function fetchConsumibles() {
    const { data, error } = await supabase
      .from('consumible')
      .select('codigo_barra, nombre');
    if (error) console.error('Error cargando consumibles:', error);
    else consumibles = data;
  }

  // --- FUNCIONES DEL FORMULARIO (HU-010) ---
  function addItem() {
    items = [...items, { consumible_id: '', cantidad_requerida: 1 }];
  }

  function removeItem(index) {
    items = items.filter((_, i) => i !== index);
  }
  
  function resetForm() {
    selectedOperario = '';
    selectedOrden = '';
    selectedFecha = new Date().toISOString().split('T')[0];
    items = [{ consumible_id: '', cantidad_requerida: 1 }];
    editingId = null;
    consumibleSearchTerm = ''; // También limpiamos la búsqueda
  }

  // --- FUNCIÓN PRINCIPAL DE ENVÍO (HU-014) ---
  async function handleSubmit() {
    if (!selectedOperario || !selectedOrden || items.length === 0) {
      alert('Por favor complete todos los campos de la solicitud.');
      return;
    }
    
    const confirmMsg = editingId 
      ? '¿Está seguro de que desea ACTUALIZAR esta solicitud?'
      : '¿Está seguro de que desea REGISTRAR esta nueva solicitud?';
      
    if (!confirm(confirmMsg)) {
      return;
    }

    const especificacionItems = items
      .filter(item => item.consumible_id && item.cantidad_requerida > 0)
      .map(item => ({
        consumible_id: item.consumible_id,
        cantidad_requerida: item.cantidad_requerida,
      }));

    if (especificacionItems.length === 0) {
      alert('Debe añadir al menos un consumible válido.');
      return;
    }

    if (editingId) {
      // --- EDITANDO ---
      const { error: updateError } = await supabase
        .from('solicitud_consumible')
        .update({
          orden_id: selectedOrden,
          operario_id: selectedOperario,
          fecha_entrega: selectedFecha,
        })
        .eq('codigo_solicitud', editingId);

      if (updateError) {
        console.error('Error actualizando la solicitud:', updateError);
        alert('Error al actualizar.');
        return;
      }

      const { error: deleteError } = await supabase
        .from('especificacion')
        .delete()
        .eq('solicitud_id', editingId);
        
      if (deleteError) {
        console.error('Error borrando especificaciones antiguas:', deleteError);
      }
      
      const { error: insertError } = await supabase
        .from('especificacion')
        .insert(especificacionItems.map(item => ({
          ...item,
          solicitud_id: editingId
        })));
        
      if (insertError) {
        console.error('Error insertando nuevas especificaciones:', insertError);
        alert('Error al actualizar los consumibles.');
        return;
      }
      
      alert('¡Solicitud ACTUALIZADA con éxito!');

    } else {
      // --- CREANDO ---
      const { data: solicitudData, error: solicitudError } = await supabase
        .from('solicitud_consumible')
        .insert({
          orden_id: selectedOrden,
          operario_id: selectedOperario,
          fecha_entrega: selectedFecha,
        })
        .select('codigo_solicitud')
        .single();

      if (solicitudError) {
        console.error('Error creando la solicitud:', solicitudError);
        alert('Error creando la solicitud.');
        return;
      }

      const nuevaSolicitudId = solicitudData.codigo_solicitud;

      const { error: especificacionError } = await supabase
        .from('especificacion')
        .insert(especificacionItems.map(item => ({
          ...item,
          solicitud_id: nuevaSolicitudId
        })));

      if (especificacionError) {
        console.error('Error insertando especificaciones:', especificacionError);
        alert('Error guardando los consumibles de la solicitud.');
        return;
      }

      alert('¡Solicitud REGISTRADA con éxito!');
    }

    resetForm();
    await fetchSolicitudes();
  }
  
  // --- HU-014: Funciones de Edición ---
  async function handleEdit(solicitud) {
    editingId = solicitud.codigo;
    selectedOperario = solicitud.operario_id;
    selectedOrden = solicitud.orden_id;
    selectedFecha = solicitud.fecha;
    
    if (solicitud.expanded) {
      solicitud.expanded = false;
    }
    
    const { data, error } = await supabase
      .from('especificacion')
      .select('consumible_id, cantidad_requerida')
      .eq('solicitud_id', solicitud.codigo);
      
    if (error) {
      alert('Error al cargar los items para editar.');
      items = [{ consumible_id: '', cantidad_requerida: 1 }];
    } else {
      // Si no hay items, Svelte no actualiza bien el array vacío.
      // Nos aseguramos de que haya al menos una fila vacía si no hay datos.
      items = data.length > 0 ? data : [{ consumible_id: '', cantidad_requerida: 1 }];
    }
    
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }
  
  function cancelEdit() {
    resetForm();
  }

  // --- HU-012: Funciones del Filtro de Fecha ---
  async function handleFilter() {
    if (!filtroFecha) {
      alert('Por favor seleccione una fecha para filtrar.');
      return;
    }
    loading = true;
    
    const { data, error } = await supabase
      .from('solicitud_consumible')
      .select('codigo_solicitud, orden_id, fecha_entrega, operario_id, usuario ( nombre )')
      .eq('fecha_entrega', filtroFecha); 

    if (error) {
      console.error('Error filtrando solicitudes:', error);
    } else {
      solicitudes = data.map((solicitud) => ({
        codigo: solicitud.codigo_solicitud,
        orden_id: solicitud.orden_id,
        operario: solicitud.usuario.nombre,
        operario_id: solicitud.operario_id,
        fecha: solicitud.fecha_entrega,
        expanded: false,
        loadingDetails: false,
        details: []
      }));
    }
    loading = false;
  }

  async function clearFilter() {
    filtroFecha = '';
    loading = true;
    await fetchSolicitudes();
    loading = false;
  }
  
  // --- ¡CAMBIO 2 (Firma de la función)! ---
  // --- HU-012: Función para filtro de consumibles ---
  function getFilteredConsumibles(searchTerm, list) {
    // Si no hay término de búsqueda, devuelve la lista completa
    if (!searchTerm) {
      return list; // 'list' es el array 'consumibles'
    }
    
    const lowerSearch = searchTerm.toLowerCase();
    
    // --- ¡CAMBIO 3 (Lógica de filtro)! ---
    // Ahora busca por nombre O por ID
    return list.filter(c => 
      c.nombre.toLowerCase().includes(lowerSearch) || 
      c.codigo_barra.toString().includes(searchTerm) // searchTerm original, por si buscan '2000'
    );
  }
  
  // --- HU-014: Función para Ver Detalles (Fila expandible) ---
  async function toggleDetalles(clickedSolicitud) {
    if (editingId) return;
    
    clickedSolicitud.expanded = !clickedSolicitud.expanded;

    if (clickedSolicitud.expanded && clickedSolicitud.details.length === 0) {
      clickedSolicitud.loadingDetails = true;
      const { data, error } = await supabase
        .from('especificacion')
        .select(`cantidad_requerida, consumible ( nombre )`)
        .eq('solicitud_id', clickedSolicitud.codigo);

      if (error) {
        console.error('Error cargando detalles:', error);
        clickedSolicitud.expanded = false;
      } else {
        clickedSolicitud.details = data;
      }
      clickedSolicitud.loadingDetails = false;
    }
    solicitudes = [...solicitudes];
  }
</script>

<main>
  <section class="card form-section">

            </select>

            </button>
          </div>
        {/each}

        <button type="button" class="btn-add" on:click={addItem}>
          + Añadir Consumible
        </button>
      </fieldset>

      <div class="form-actions">
        <button type="submit" class="btn-submit">
          {editingId ? 'Actualizar Solicitud' : 'Enviar Solicitud'}
        </button>
        
        {#if editingId}
          <button type="button" class="btn-cancel" on:click={cancelEdit}>
            Cancelar Edición
          </button>
        {/if}
      </div>
    </form>
  </section>

  <section class="card table-section">
    <h2>Historial de Solicitudes</h2>

    <div class="filter-bar">
      <label for="filtro-fecha">Filtrar por fecha:</label>
      <input type="date" id="filtro-fecha" bind:value={filtroFecha} />
      <button type="button" class="btn-filter" on:click={handleFilter}>
        Filtrar
      </button>
      <button type="button" class="btn-clear" on:click={clearFilter}>
        Limpiar Filtro
      </button>
    </div>

    <table>
      <thead>
        <tr>
          <th>Cód. Solicitud</th>
          <th>Orden Producción</th>
          <th>Operario</th>
          <th>Fecha Entrega</th>
          <th>Acciones</th>
        </tr>
      </thead>
      
      {#if loading}
        <tbody>
          <tr><td colspan="5">Cargando datos desde Supabase...</td></tr>
        </tbody>
      {:else}
        {#each solicitudes as solicitud (solicitud.codigo)}
          <tbody class="solicitud-group">
            <tr>
              <td>{solicitud.codigo}</td>
              <td>{solicitud.orden_id}</td>
              <td>{solicitud.operario}</td>
              <td>{solicitud.fecha}</td>
              
              <td class="action-buttons">
                <button 
                  type="button" 
                  class="btn-details" 
                  on:click={() => toggleDetalles(solicitud)}
                  disabled={editingId}
                >
                  {solicitud.expanded ? 'Ocultar' : 'Ver Detalles'}
                </button>
                <button 
                  type="button" 
                  class="btn-edit" 
                  on:click={() => handleEdit(solicitud)}
                  disabled={editingId === solicitud.codigo}
                >
                  Editar
                </button>
              </td>
            </tr>

            {#if solicitud.expanded}
              <tr class="detalle-row">
                <td colspan="5">
                  <div class="detalle-contenido">
                    {#if solicitud.loadingDetails}
                      <p>Cargando detalles...</p>
                    {:else if solicitud.details.length === 0}
                      <p>Esta solicitud no tiene consumibles registrados.</p>
                    {:else}
                      <h5>Consumibles Solicitados:</h5>
                      <ul class="detalle-lista">
                        {#each solicitud.details as item}
                          <li>
                            <span class="item-nombre">{item.consumible.nombre}</span>
                            <span class="item-cantidad">Cantidad: {item.cantidad_requerida}</span>
                          </li>
                        {/each}
                      </ul>
                    {/if}
                  </div>
                </td>
              </tr>
            {/if}
          </tbody>
        {/each}
      {/if}
    </table>
  </section>
</main>
